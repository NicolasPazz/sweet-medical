// ─── Helper: Renderizar estrellas ─────────────────────────────────────────────
function Stars({ rating }) {
  const full = Math.floor(rating);
  return (
    <span className="stars">
      {"★".repeat(full)}{"☆".repeat(5 - full)}
    </span>
  );
}

// ─── Mini Calendar ────────────────────────────────────────────────────────────
function MiniCalendar({ selectedDay, onSelect }) {
  const [viewMonth, setViewMonth] = useState(9); // octubre = 9
  const [viewYear]  = useState(2024);

  const firstDay = new Date(viewYear, viewMonth, 1).getDay(); // 0=dom
  const daysInMonth = new Date(viewYear, viewMonth + 1, 0).getDate();
  // offset: monday-first
  const offset = (firstDay + 6) % 7;

  const cells = [];
  for (let i = 0; i < offset; i++) cells.push(null);
  for (let d = 1; d <= daysInMonth; d++) cells.push(d);

  return (
    <div>
      <div className="cal-header">
        <button className="cal-nav-btn" onClick={() => setViewMonth(m => m - 1)}>‹</button>
        <span className="cal-month">{MONTHS[viewMonth]} {viewYear}</span>
        <button className="cal-nav-btn" onClick={() => setViewMonth(m => m + 1)}>›</button>
      </div>
      <div className="cal-week-header">
        {WEEKDAYS.map(d => <div className="cal-week-day" key={d}>{d}</div>)}
      </div>
      <div className="cal-days">
        {cells.map((d, i) => {
          if (!d) return <div key={`e${i}`} />;
          const available = AVAILABLE_DAYS.includes(d);
          const isSelected = d === selectedDay;
          const isToday = d === 14;
          return (
            <div
              key={d}
              onClick={() => available && onSelect(d)}
              className={[
                "cal-day",
                isToday ? "today" : "",
                isSelected ? "selected" : "",
                available ? "has-slots" : "disabled",
              ].join(" ").trim()}
            >
              {d}
            </div>
          );
        })}
      </div>
      <div style={{ marginTop: 14, display: "flex", gap: 12, fontSize: 11, color: "var(--neutral-500)" }}>
        <span style={{ display: "flex", alignItems: "center", gap: 5 }}>
          <span style={{ width: 8, height: 8, borderRadius: "50%", background: "var(--brand-light)", display: "inline-block" }} />
          Turnos disponibles
        </span>
        <span style={{ display: "flex", alignItems: "center", gap: 5 }}>
          <span style={{ width: 8, height: 8, borderRadius: "50%", background: "var(--neutral-300)", display: "inline-block" }} />
          Sin disponibilidad
        </span>
      </div>
    </div>
  );
}

// ─── Sidebar con preseleccionados ─────────────────────────────────────────────
function PreselSidebar({ items, onRemove, onConfirm }) {
  return (
    <div className="presel-sidebar">
      <div className="presel-title">
        <span>📋</span> Preseleccionados
        {items.length > 0 && (
          <span style={{ marginLeft: "auto", background: "var(--brand)", color: "#fff", borderRadius: "12px", padding: "2px 8px", fontSize: 11 }}>
            {items.length}
          </span>
        )}
      </div>

      {items.length === 0 && (
        <p style={{ fontSize: 12, color: "var(--neutral-500)", lineHeight: 1.5 }}>
          Aún no hay turnos preseleccionados. Seleccioná un horario para agregarlo aquí.
        </p>
      )}

      {items.map((item, idx) => (
        <div className="presel-item" key={idx}>
          <span className="presel-num">{idx + 1}</span>
          <span className="presel-info">
            <strong>{item.specialty}</strong> – {item.doctor}<br />
            {item.day} Oct, {item.time}<br />
            <span style={{ color: "var(--neutral-400, #9CA3AF)" }}>({item.type}, {item.sede})</span>
          </span>
          <span className="presel-del" onClick={() => onRemove(idx)}>✕</span>
        </div>
      ))}

      {items.length > 0 && (
        <>
          <div className="divider" />
          <button className="btn btn-primary" style={{ width: "100%", justifyContent: "center" }} onClick={onConfirm}>
            Confirmar Selección ({items.length} {items.length === 1 ? "Turno" : "Turnos"})
          </button>
        </>
      )}
    </div>
  );
}

                        <button className="btn btn-ghost btn-sm" onClick={() => setStep(4)}>✎ Cambiar</button>
                      </div>
                      <div className="confirm-row">
                        <div>
                          <div className="confirm-label">Modalidad</div>
                          <div className="confirm-value">{selectedSlot?.type}</div>
                        </div>
                      </div>
                    </div>

                    {/* Selección de paciente */}
                    <div style={{ marginBottom: 20 }}>
                      <div style={{ fontSize: 14, fontWeight: 700, marginBottom: 12 }}>
                        👤 ¿Para quién es el turno?
                      </div>
                      <div className="pac-list">
                        {PATIENTS.map(pac => (
                          <div
                            key={pac.id}
                            className={`pac-card ${patient?.id === pac.id ? "selected" : ""}`}
                            onClick={() => setPatient(pac)} // aquí se cargarían coberturas reales del paciente
                          >
                            <div className="pac-avatar">{pac.initials}</div>
                            <div>
                              <div className="pac-name">{pac.name}</div>
                              <div className="pac-sub">{pac.sub}</div>
                            </div>
                            {patient?.id === pac.id && <span style={{ marginLeft: "auto", color: "var(--brand)", fontSize: 18 }}>✓</span>}
                          </div>
                        ))}
                      </div>
                    </div>

                    <div className="btn-row">
                      <button className="btn btn-ghost" onClick={() => setStep(4)}>← Volver</button>
                      <button
                        className="btn btn-success"
                        disabled={!patient}
                        style={{ opacity: patient ? 1 : .4, cursor: patient ? "pointer" : "not-allowed" }}
                        onClick={() => {
                          // Aquí iría la llamada real al backend para confirmar el turno
                          const newItem = { specialty: specialty?.name, doctor: doctor?.name, day: selectedDay, time: selectedSlot?.time, type: selectedSlot?.type, sede: doctor?.sede };
                          setPreselected([newItem]);
                          setConfirmed(true);
                        }}
                      >
                        🎉 Confirmar Turno
                      </button>
                    </div>
                  </div>
                </div>
              </div>
            )}

          </div>
        </div>
      </div>
    </>
  );
}
